\section{Contexte et positionnement}
% ════════════════════════════════════════════════════════════

\begin{frame}{Contexte et fondements théoriques}
  \begin{itemize}[<+->]
    \item Effet photovoltaïque \& modèles physiques
    \item Composants d'un système PV
    \item Taxonomie des défauts PV
    \item Limites des méthodes de détection classiques
  \end{itemize}
\end{frame}

\begin{frame}{Pourquoi la modalité électrique + météo}
  \begin{itemize}[<+->]
    \item \textbf{Imagerie thermique} : coût de 0,5 à 3~€/module, acquisition événementielle (drone) et donc pas de continuité temporelle.
    \item \textbf{Courbe I-V} : mesure discrète, interrompt la production, sensible aux variations d'irradiance pendant le balayage.
    \item \alert{\textbf{Électrique + météo}} : Suivi à haute résolution temporelle, déjà collectées par l'infrastructure SCADA existante.
  \end{itemize}
  \vfill
  \onslide<4->{\footnotesize C'est la seule modalité compatible avec une surveillance \alert{continue et near real-time}, condition nécessaire pour une \alert{détection précoce} des défauts.}
\end{frame}

\begin{frame}{État de l'art : panorama par famille de méthodes}
  \footnotesize
  Méthodes ML/DL appliquées aux séries temporelles multivariées électrique + météo en contexte PV, organisées par famille :
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \footnotesize
      \textbf{Hors ligne -- Machine Learning}
      \begin{itemize}
        \item \textbf{Noyau} (SVM, OC-SVM) -- Suliman et al. (2024) : SVM multi-noyau vs XGBoost adaptatif, 87,56\%
        \item \textbf{Ensembles} (RF, Isolation Forest) -- Amiri et al. (2024) : Random Forest, 99,4\% (détection + diagnostic)
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \footnotesize
      \textbf{Hors ligne -- Deep Learning}
      \begin{itemize}
        \item \textbf{Séquentiel} (LSTM, SSM, Transformer) -- Hajji et al. (2023) : BiLSTM, $\approx$100\% (simulé)
        \item \textbf{Reconstruction} (AE, VAE) -- Seghiour et al. (2023) : autoencodeur, 99,9\%
      \end{itemize}
    \end{column}
  \end{columns}
  \vfill
  \textbf{En ligne (streaming)} -- changepoint bayésien (BOCD), manifold régularisé, passive-aggressive : peu exploré côté PV malgré la compatibilité naturelle avec l'edge ; repris comme un des détecteurs évalués dans ce mémoire.
\end{frame}

\begin{frame}{Synthèse de la littérature}
  Quatre facettes d'une même limite structurelle, pas une liste de manques isolés :
  \begin{enumerate}[<+->]
    \item \textbf{Dominance de la simulation} -- la majorité des travaux valident sur données synthétiques (MATLAB/Simulink), pas sur le bruit et la non-linéarité du terrain
    \item \textbf{Ambiguïté des défauts à faible signal} -- les détecteurs par reconstruction peinent sur les dérives lentes (Dégradation) qui se confondent avec la variabilité normale
    \item \textbf{Déploiement embarqué rarement mesuré} -- l'efficacité est presque toujours exprimée en proxys (nb.\ paramètres, FLOPs) sur poste de travail, pas en latence mesurée sur device cible
    \item \textbf{Drift non surveillé} -- les campagnes de mesure couvrent quelques semaines, jamais un cycle annuel complet ; le glissement saisonnier n'est ni détecté ni signalé à l'opérateur
  \end{enumerate}
  \vfill
  \onslide<5->{\centering\alert{\textbf{Ce mémoire répond aux quatre, sur le même système, avec les mêmes données.}}}
\end{frame}

\begin{frame}{Feature engineering physique}
  \footnotesize
  \texttt{plus\_physics} : 4 familles de features physiques ($\approx$17 features), motivées par l'EDA et conçues pour rester stables sur toute la plage de fonctionnement.
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{1. Canaux normalisés par irradiance} -- isole le résidu lié au défaut de la grande amplitude diurne :
      \[
        \tilde{P}_{s} = \frac{P_{s}}{\max(G, G_{\min})}
      \]
      \textbf{2. Déséquilibre inter-strings} -- isole les défauts localisés invisibles à l'échelle de la centrale :
      \[
        \Delta P = \frac{P_1 - P_2}{P_1 + P_2 + \varepsilon}
      \]
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{3. Puissance corrigée en température} -- retire la composante thermique attendue :
      \[
        \tilde{P}_{\mathrm{corr}} = \frac{P_{\mathrm{dc}}}{(1+\gamma(T-T_{\mathrm{ref}}))\cdot G}
      \]
      \textbf{4. Dérivées temporelles d'ordre 1} -- expose les transitoires rapides (courts-circuits, circuits ouverts) :
      \[
        \delta P(t) = P(t) - P(t-1)
      \]
    \end{column}
  \end{columns}
  \vfill
  Sélection \alert{mRMR (critère MID)} appliquée sous contrainte train-only, pour retenir les features non redondantes les plus informatives sans fuite vers val/test.
\end{frame}

\begin{frame}{Outillage MLOps}
  \footnotesize
  Au-delà de la capacité conceptuelle, la chaîne expérimentale repose sur un outillage d'ingénierie reproductible :
  \begin{itemize}
    \item \textbf{DVC} -- versioning des données et artefacts par hash de contenu ; tout changement amont invalide et régénère automatiquement les étapes en aval
    \item \textbf{Pipeline DAG reproductible} -- 5 étages (\texttt{ingest} $\to$ \texttt{split} $\to$ \texttt{preprocess} $\to$ \texttt{featurize} $\to$ \texttt{train}), ré-exécutable à l'identique
    \item \textbf{MLflow} -- traçabilité de chaque run (dataset, profil de features, hyperparamètres, métriques val/test), hébergé sur DagsHub
  \end{itemize}
\end{frame}

  \onslide<7->{\scriptsize\alert{La Dégradation est un défaut de contact à haute résistance, injecté de façon contrôlée} (et non une dérive d'usure naturelle) : chaque mesure isolée reste proche de la normale, sans événement net. PC-Flow score par \emph{probabilité conditionnelle} (irradiance, température), pas par magnitude d'erreur -- un écart discret par rapport à la variété physique attendue tombe dans la queue de faible probabilité même quand chaque signal a l'air ordinaire.}
\end{frame}

\begin{frame}{PC-Flow --- principe}
  Un \alert{normalizing flow} : un transport bijectif $f_\theta : \mathbb{R}^n \to \mathbb{R}^n$, $z = f_\theta^{-1}(x)$, exactement inversible.
  \begin{itemize}[<+->]
    \item Contrairement à un VAE (borne inférieure sur la vraisemblance), le flow donne une \alert{vraisemblance exacte} via le changement de variables :
    \[
      \log p(x;\theta) = \log p_Z\bigl(f_\theta^{-1}(x)\bigr) + \log\left|\det\frac{\partial f_\theta^{-1}(x)}{\partial x}\right|
    \]
    \item Architecture : conditional \texttt{RealNVP}, coupling layers affines, contexte $c = [\text{irradiance}, T_{\text{PV}}]$ injecté à chaque bloc
    \item Score d'anomalie = $-\log p(x \mid c;\theta)$ -- pas de feature engineering manuel requis
    \item \alert{Aucun travail antérieur} n'avait appliqué un normalizing flow à la détection de défauts PV
  \end{itemize}
\end{frame}

\begin{frame}{PC-Flow --- intuition géométrique}
  \begin{center}
    \onslide<1->{\fbox{\parbox{0.85\textwidth}{\centering\vspace{0.7cm}\textit{[Figure : schéma architecture PC-Flow -- coupling layers + conditionnement physique]}\vspace{0.7cm}}}}
  \end{center}
  \onslide<2->{\scriptsize Le flow apprend une densité \emph{conditionnée} $p(x\mid c)$ : il factorise la façon dont irradiance et température pilotent normalement les signaux électriques. Une dérive par rapport à cette variété physique -- pas le niveau absolu du signal -- est ce qui est pénalisé.}
\end{frame}

\begin{frame}{Annexe B --- Justification du macro PR-AUC}
  \footnotesize
  \[
    \mathrm{PR\text{-}AUC}_{\mathrm{macro}} = \frac{1}{K}\sum_{k=1}^{K} \mathrm{PR\text{-}AUC}_k, \qquad K = 4
  \]
  \begin{itemize}[<+->]
    \item Le binary PR-AUC pondère implicitement par la fréquence -- Shadowing (35,7\% des lignes en panne) domine le calcul
    \item Le macro PR-AUC donne un poids \alert{égal} à chaque classe, indépendamment de sa fréquence
    \item Une panne rare (Short-Circuit, 15 épisodes) compte autant qu'une panne fréquente (Shadowing, 1000 épisodes)
    \item Cohérent avec l'objectif opérationnel : un détecteur aveugle à une panne rare reste inacceptable même s'il excelle sur la panne fréquente
  \end{itemize}
\end{frame}

\begin{frame}{Annexe D --- Ablations PC-Flow}
  \footnotesize
  \begin{itemize}[<+->]
    \item \textbf{Profil de features} : macro PR-AUC se dégrade de 0,990 (7 features) à 0,955 (153 features, \texttt{plus\_rolling}) -- malédiction de la dimensionnalité ; le flow apprend déjà sa propre représentation

  \end{itemize}

\begin{frame}{Annexe E --- MCE = 0,754, explication}
  \footnotesize
  \begin{itemize}[<+->]
    \item Le MCE capture l'écart \alert{pire-cas} entre confiance prédite et exactitude observée, sur un seul bin de confiance
    \item À ne pas confondre avec l'ECE (0,019), qui moyenne cet écart sur \emph{tous} les bins, pondéré par leur population
    \item Le bin concerné est un bin de très haute confiance et de très faible population (\og tail-bin effect \fg) -- quelques échantillons mal classés y suffisent à produire un grand écart relatif
    \item Log-loss (0,059) et Brier (0,008) confirment que la qualité \emph{globale} de calibration reste excellente
  \end{itemize}
\end{frame}

\begin{frame}{Annexe F --- Protocole de Drift opérationnel}
  \footnotesize
  Test \texttt{Page-Hinkley} bidirectionnel sur les statistiques journalières du régime opératoire $(\mathrm{irr}, T_{PV})$ :
  \[
    M_t = \sum_{i=1}^{t}(x_i - \bar{x} - \delta), \quad \mathrm{PH}_t = M_t - \min_{j\le t} M_j, \quad \text{drift} = \mathbf{1}[\mathrm{PH}_t > \lambda]
  \]
  \begin{itemize}
    \item Agrégation \alert{journalière} : distingue le drift saisonnier persistant du cycle diurne attendu
    \item Stade \textbf{advisory} après 7 jours consécutifs de glissement
    \item Stade \textbf{action-ready} après 14 jours consécutifs
    \item Surveille le \emph{régime}, indépendamment du score du modèle -- évite la circularité (cf.\ feedback circularité \& gates)
  \end{itemize}
\end{frame}
